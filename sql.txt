/* =========================================================
   6) tickets
   ========================================================= */
IF OBJECT_ID(N'dbo.tickets', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.tickets (
        id UNIQUEIDENTIFIER NOT NULL CONSTRAINT PK_tickets PRIMARY KEY DEFAULT NEWID(),
        ticket_code NVARCHAR(64) NOT NULL,
        title NVARCHAR(255) NOT NULL,
        description NVARCHAR(MAX) NOT NULL,
        category NVARCHAR(20) NOT NULL,
        priority NVARCHAR(20) NOT NULL,
        status NVARCHAR(20) NOT NULL CONSTRAINT DF_tickets_status DEFAULT (N'NEW'),
        emitter_department_id INT NOT NULL,
        target_department_id INT NOT NULL,
        creator_id UNIQUEIDENTIFIER NOT NULL,
        assignee_id UNIQUEIDENTIFIER NULL,
        resolved_at DATETIME2(3) NULL,
        closed_at DATETIME2(3) NULL,
        created_at DATETIME2(3) NOT NULL CONSTRAINT DF_tickets_created_at DEFAULT (SYSUTCDATETIME()),
        updated_at DATETIME2(3) NOT NULL CONSTRAINT DF_tickets_updated_at DEFAULT (SYSUTCDATETIME()),

        CONSTRAINT UQ_tickets_ticket_code UNIQUE (ticket_code),
        CONSTRAINT CK_tickets_category CHECK (category IN (N'REQUEST',N'INCIDENT',N'ANOMALY',N'VALIDATION')),
        CONSTRAINT CK_tickets_priority CHECK (priority IN (N'LOW',N'MEDIUM',N'HIGH',N'CRITICAL')),
        CONSTRAINT CK_tickets_status CHECK (status IN (N'NEW',N'IN_PROGRESS',N'ON_HOLD',N'RESOLVED',N'CLOSED')),

        CONSTRAINT FK_tickets_emitter_department FOREIGN KEY (emitter_department_id) REFERENCES dbo.departments(id),
        CONSTRAINT FK_tickets_target_department FOREIGN KEY (target_department_id) REFERENCES dbo.departments(id),
        CONSTRAINT FK_tickets_creator FOREIGN KEY (creator_id) REFERENCES dbo.users(id),
        CONSTRAINT FK_tickets_assignee FOREIGN KEY (assignee_id) REFERENCES dbo.users(id)
    );
END
GO

/* =========================================================
   13) seed de base (departments + SLA)
   ========================================================= */
MERGE dbo.departments AS t
USING (VALUES
    (N'RH',  N'Ressources Humaines'),
    (N'FIN', N'Finance'),
    (N'REC', N'Recouvrement'),
    (N'IT',  N'Informatique'),
    (N'COM', N'Commercialisation'),
    (N'OPS', N'Operations'),
    (N'DIR', N'Direction')
) AS s(code, name)
ON t.code = s.code
WHEN NOT MATCHED THEN
    INSERT (code, name, is_active, created_at, updated_at)
    VALUES (s.code, s.name, 1, SYSUTCDATETIME(), SYSUTCDATETIME());
GO

MERGE dbo.sla_policies AS t
USING (VALUES
    (N'LOW',      960, 4800),
    (N'MEDIUM',   480, 2400),
    (N'HIGH',     240,  960),
    (N'CRITICAL',  60,  480)
) AS s(priority, first_response_minutes, resolution_minutes)
ON t.priority = s.priority
WHEN NOT MATCHED THEN
    INSERT (priority, first_response_minutes, resolution_minutes, is_active, created_at, updated_at)
    VALUES (s.priority, s.first_response_minutes, s.resolution_minutes, 1, SYSUTCDATETIME(), SYSUTCDATETIME());
GO
